## 1. 核心代码流程分析

### 1.1 主训练脚本结构 (`async_sac_state_sim.py`)

```python
# ==================== 第一部分：导入和配置 ====================
from agentlace.data.data_store import QueuedDataStore
from agentlace.trainer import TrainerClient, TrainerServer
from serl_launcher.utils.launcher import (
    make_sac_agent,
    make_trainer_config,
    make_replay_buffer,
)

# 定义命令行参数
FLAGS = flags.FLAGS
flags.DEFINE_float('utd_ratio', 1.0, 'Update-to-data ratio.')
flags.DEFINE_integer("batch_size", 256, "Batch size.")
# ... 更多参数

# ==================== 第二部分：Actor 函数 ====================
def actor(agent: SACAgent, data_store, env, sampling_rng):
    """
    Actor 循环：与环境交互，收集数据
    """
    # 1. 创建 TrainerClient，连接到 Learner
    client = TrainerClient(
        "actor_env",
        FLAGS.ip,
        make_trainer_config(),
        data_store,
        wait_for_server=True,  # 等待 Learner 启动
    )
    
    # 2. 定义参数更新回调函数
    def update_params(params):
        nonlocal agent
        agent = agent.replace(state=agent.state.replace(params=params))
    
    client.recv_network_callback(update_params)
    
    # 3. 主循环
    obs, _ = env.reset()
    for step in range(FLAGS.max_steps):
        # 3.1 采样动作
        if step < FLAGS.random_steps:
            actions = env.action_space.sample()  # 随机探索
        else:
            actions = agent.sample_actions(obs, deterministic=False)
        
        # 3.2 执行动作
        next_obs, reward, done, truncated, info = env.step(actions)
        
        # 3.3 存储经验到 DataStore
        data_store.insert({
            'observations': obs,
            'actions': actions,
            'next_observations': next_obs,
            'rewards': reward,
            'masks': 1.0 - done,
            'dones': done or truncated,
        })
        
        # 3.4 重置环境（如果需要）
        if done or truncated:
            obs, _ = env.reset()
        else:
            obs = next_obs
        
        # 3.5 定期评估
        if step % FLAGS.eval_period == 0:
            evaluate(agent, eval_env, num_episodes=5)

# ==================== 第三部分：Learner 函数 ====================
def learner(rng, agent):
    """
    Learner 循环：接收数据，训练策略
    """
    # 1. 创建 DataStore 和 TrainerServer
    data_store = QueuedDataStore(capacity=50000)
    server = TrainerServer(
        make_trainer_config(),
        data_store,
        wait_for_server=True,
    )
    
    # 2. 创建 Replay Buffer
    replay_buffer = make_replay_buffer(
        env,
        capacity=FLAGS.replay_buffer_capacity,
    )
    
    # 3. 填充 Replay Buffer
    while replay_buffer.size < FLAGS.training_starts:
        batch = data_store.recv()  # 从 Actor 接收数据
        replay_buffer.insert(batch)
    
    # 4. 训练循环
    for step in range(FLAGS.max_steps):
        # 4.1 从 Buffer 采样
        batch = replay_buffer.sample(
            batch_size=FLAGS.batch_size * FLAGS.utd_ratio
        )
        
        # 4.2 更新 Agent
        agent, update_info = agent.update(batch, FLAGS.utd_ratio)
        
        # 4.3 同步参数到 Actor
        if step % FLAGS.steps_per_update == 0:
            server.publish_network(agent.state.params)
        
        # 4.4 记录日志
        if step % FLAGS.log_period == 0:
            wandb.log(update_info, step=step)
        
        # 4.5 保存检查点
        if step % FLAGS.checkpoint_period == 0:
            checkpoints.save_checkpoint(
                FLAGS.checkpoint_path,
                agent.state,
                step=step,
            )

# ==================== 第四部分：主函数 ====================
def main(_):
    # 1. 初始化环境
    env = gym.make(FLAGS.env)
    env = RecordEpisodeStatistics(env)
    
    # 2. 初始化 Agent
    rng = jax.random.PRNGKey(FLAGS.seed)
    agent = make_sac_agent(
        sample_obs=env.observation_space.sample(),
        sample_action=env.action_space.sample(),
    )
    
    # 3. 启动 Actor 或 Learner
    if FLAGS.learner:
        learner(rng, agent)
    elif FLAGS.actor:
        data_store = QueuedDataStore(capacity=50000)
        actor(agent, data_store, env, rng)
```

### 2.1 `make_sac_agent()` - Agent 创建器

**位置**: `serl_launcher/utils/launcher.py`

```python
def make_sac_agent(
    seed: int,
    sample_obs: np.ndarray,
    sample_action: np.ndarray,
    image_keys: Tuple[str] = ("image",),
    encoder_type: str = "resnet-pretrained",
    **kwargs
) -> SACAgent:
    """
    创建 SAC Agent
    
    参数:
        seed: 随机种子
        sample_obs: 样本观测（用于推断形状）
        sample_action: 样本动作（用于推断形状）
        image_keys: 图像键名（用于视觉观测）
        encoder_type: 编码器类型
        
    返回:
        SACAgent: 初始化好的 Agent
    """
    
    # 1. 初始化随机数生成器
    rng = jax.random.PRNGKey(seed)
    rng, actor_key, critic_key, temp_key = jax.random.split(rng, 4)
    
    # 2. 创建 Actor 网络
    if encoder_type == "none":
        # 状态观测：使用 MLP
        actor_def = Policy(
            hidden_dims=(256, 256),
            action_dim=sample_action.shape[-1],
        )
    else:
        # 图像观测：使用 CNN + MLP
        encoder_def = encoders[encoder_type]
        actor_def = PolicyWithEncoder(
            encoder=encoder_def,
            network=Policy(hidden_dims=(256, 256)),
        )
    
    # 3. 创建 Critic 网络（双 Q 网络）
    critic_def = DoubleCritic(
        hidden_dims=(256, 256),
    )
    
    # 4. 创建温度参数
    temp_def = Temperature(init_temperature=1.0)
    
    # 5. 初始化参数
    actor_params = actor_def.init(actor_key, sample_obs)['params']
    critic_params = critic_def.init(
        critic_key, sample_obs, sample_action
    )['params']
    temp_params = temp_def.init(temp_key)['params']
    
    # 6. 创建 Agent
    return SACAgent.create(
        actor_def=actor_def,
        critic_def=critic_def,
        temp_def=temp_def,
        actor_params=actor_params,
        critic_params=critic_params,
        temp_params=temp_params,
        **kwargs
    )
```

### 2.2 `SACAgent` 类详解

**位置**: `serl_launcher/agents/continuous/sac.py`

```python
@partial(jax.jit, static_argnames=['self', 'utd_ratio'])
def update(self, batch: Batch, utd_ratio: int):
    """
    SAC 更新函数
    
    参数:
        batch: 从 Replay Buffer 采样的数据
        utd_ratio: Update-to-Data 比率
        
    返回:
        (new_agent, info_dict): 更新后的 agent 和训练信息
    """
    
    # ========== 第一步：更新 Critic ==========
    def critic_loss_fn(critic_params):
        # 1. 计算当前 Q 值
        q1, q2 = self.critic(critic_params, batch['observations'], batch['actions'])
        
        # 2. 计算目标 Q 值
        # 2.1 采样下一个动作
        next_actions, next_log_probs = self.actor.sample(
            batch['next_observations']
        )
        
        # 2.2 计算 target Q 值（使用 target network）
        next_q1, next_q2 = self.target_critic(
            batch['next_observations'], next_actions
        )
        next_q = jnp.minimum(next_q1, next_q2)  # 双 Q 网络
        
        # 2.3 TD Target = r + γ * (min(Q') - α*log π)
        target_q = batch['rewards'] + self.discount * batch['masks'] * (
            next_q - self.temp * next_log_probs
        )
        
        # 3. 计算 TD Error
        critic_loss = ((q1 - target_q)**2 + (q2 - target_q)**2).mean()
        
        return critic_loss, {
            'critic_loss': critic_loss,
            'q1': q1.mean(),
            'q2': q2.mean(),
        }
    
    # 更新 Critic 网络 (utd_ratio * critic_actor_ratio 次)
    for _ in range(utd_ratio * self.critic_actor_ratio):
        new_critic_params, critic_info = jax.grad(
            critic_loss_fn, has_aux=True
        )(self.critic_params)
        
        self.critic_params = self.critic_optimizer.update(
            new_critic_params, self.critic_params
        )
    
    # ========== 第二步：更新 Actor ==========
    def actor_loss_fn(actor_params):
        # 1. 采样动作
        actions, log_probs = self.actor.sample(batch['observations'])
        
        # 2. 计算 Q 值
        q1, q2 = self.critic(batch['observations'], actions)
        q = jnp.minimum(q1, q2)
        
        # 3. Actor 损失 = E[α*log π - Q]
        actor_loss = (self.temp * log_probs - q).mean()
        
        return actor_loss, {
            'actor_loss': actor_loss,
            'entropy': -log_probs.mean(),
        }
    
    # 更新 Actor 网络 (utd_ratio 次)
    for _ in range(utd_ratio):
        new_actor_params, actor_info = jax.grad(
            actor_loss_fn, has_aux=True
        )(self.actor_params)
        
        self.actor_params = self.actor_optimizer.update(
            new_actor_params, self.actor_params
        )
    
    # ========== 第三步：更新温度参数 ==========
    def temp_loss_fn(temp_params):
        # 自动调整温度以匹配目标熵
        temp = self.temp_fn(temp_params)
        
        _, log_probs = self.actor.sample(batch['observations'])
        
        temp_loss = (temp * (log_probs + self.target_entropy)).mean()
        
        return temp_loss
    
    new_temp_params = jax.grad(temp_loss_fn)(self.temp_params)
    self.temp_params = self.temp_optimizer.update(
        new_temp_params, self.temp_params
    )
    
    # ========== 第四步：更新 Target Network ==========
    new_target_critic_params = jax.tree_map(
        lambda target, online: self.tau * online + (1 - self.tau) * target,
        self.target_critic_params,
        self.critic_params,
    )
    
    # 返回更新后的 Agent
    return self.replace(
        critic_params=new_critic_params,
        actor_params=new_actor_params,
        temp_params=new_temp_params,
        target_critic_params=new_target_critic_params,
    ), {**critic_info, **actor_info}
```

## 3. SAC 算法实现细节

### 3.1 Actor 网络（策略网络）

```python
class Policy(nn.Module):
    """
    高斯策略网络
    输出动作的均值和标准差
    """
    hidden_dims: Sequence[int]
    action_dim: int
    log_std_min: float = -5.0
    log_std_max: float = 2.0
    
    @nn.compact
    def __call__(self, observations: jnp.ndarray) -> tfd.Distribution:
        # 1. MLP 特征提取
        x = observations
        for dim in self.hidden_dims:
            x = nn.Dense(dim)(x)
            x = nn.relu(x)
        
        # 2. 输出均值和对数标准差
        mean = nn.Dense(self.action_dim)(x)
        log_std = nn.Dense(self.action_dim)(x)
        
        # 3. 限制 log_std 范围
        log_std = jnp.clip(log_std, self.log_std_min, self.log_std_max)
        
        # 4. 创建高斯分布
        std = jnp.exp(log_std)
        distribution = tfd.MultivariateNormalDiag(loc=mean, scale_diag=std)
        
        return distribution
    
    def sample(self, observations: jnp.ndarray, seed: jnp.ndarray):
        """
        采样动作（带重参数化技巧）
        """
        # 1. 获取分布
        dist = self(observations)
        
        # 2. 重参数化采样（可微分）
        raw_actions = dist.sample(seed=seed)
        
        # 3. Tanh 压缩到 [-1, 1]
        actions = jnp.tanh(raw_actions)
        
        # 4. 计算 log 概率（需要修正 Tanh 变换）
        log_probs = dist.log_prob(raw_actions)
        log_probs -= jnp.sum(
            jnp.log(1 - actions**2 + 1e-6), axis=-1
        )
        
        return actions, log_probs
```

### 3.2 Critic 网络（Q 网络）

```python
class DoubleCritic(nn.Module):
    """
    双 Q 网络（减少过估计）
    """
    hidden_dims: Sequence[int]
    
    @nn.compact
    def __call__(self, observations: jnp.ndarray, actions: jnp.ndarray):
        # 拼接观测和动作
        x = jnp.concatenate([observations, actions], axis=-1)
        
        # Q 网络 1
        q1 = x
        for dim in self.hidden_dims:
            q1 = nn.Dense(dim)(q1)
            q1 = nn.relu(q1)
        q1 = nn.Dense(1)(q1).squeeze(-1)
        
        # Q 网络 2（独立参数）
        q2 = x
        for dim in self.hidden_dims:
            q2 = nn.Dense(dim)(q2)
            q2 = nn.relu(q2)
        q2 = nn.Dense(1)(q2).squeeze(-1)
        
        return q1, q2
```

### 3.3 温度参数（自动调整）

```python
class Temperature(nn.Module):
    """
    可学习的温度参数
    用于自动调整探索程度
    """
    init_temperature: float = 1.0
    
    @nn.compact
    def __call__(self):
        # log(α) 是可学习参数
        log_temp = self.param(
            'log_temp',
            init_fn=lambda key: jnp.full((), jnp.log(self.init_temperature))
        )
        # 返回 α = exp(log(α))
        return jnp.exp(log_temp)
```

```python
class SERLObsWrapper(gym.ObservationWrapper):
    """
    SERL 观测包装器
    将观测转换为字典格式，便于处理多模态输入
    """
    
    def __init__(self, env, proprio_keys=None, image_keys=None):
        super().__init__(env)
        self.proprio_keys = proprio_keys or []
        self.image_keys = image_keys or []
        
    def observation(self, obs):
        """
        转换观测格式
        
        输入: obs (可能是 array 或 dict)
        输出: {
            'state': proprio_obs,
            'image': image_obs,
            'goal': goal (if available)
        }
        """
        wrapped_obs = {}
        
        # 处理状态观测
        if self.proprio_keys:
            proprio = []
            for key in self.proprio_keys:
                proprio.append(obs[key])
            wrapped_obs['state'] = np.concatenate(proprio, axis=-1)
        elif isinstance(obs, np.ndarray):
            wrapped_obs['state'] = obs
        
        # 处理图像观测
        if self.image_keys:
            images = []
            for key in self.image_keys:
                img = obs[key]
                # 归一化到 [0, 1]
                if img.dtype == np.uint8:
                    img = img.astype(np.float32) / 255.0
                images.append(img)
            
            # 多相机拼接
            if len(images) > 1:
                wrapped_obs['image'] = np.concatenate(images, axis=-1)
            else:
                wrapped_obs['image'] = images[0]
        
        return wrapped_obs
```

### 4.2 Chunking 包装器（多步动作）

```python
# serl_launcher/wrappers/chunking.py

class ChunkingWrapper(gym.Wrapper):
    """
    动作分块包装器
    允许策略输出多步动作序列
    """
    
    def __init__(self, env, obs_horizon=2, act_exec_horizon=None):
        """
        参数:
            obs_horizon: 观测历史长度
            act_exec_horizon: 执行动作的步数
        """
        super().__init__(env)
        self.obs_horizon = obs_horizon
        self.act_exec_horizon = act_exec_horizon or 1
        
        # 观测缓冲区
        self.obs_buffer = deque(maxlen=obs_horizon)
        
    def step(self, action):
        """
        执行多步动作
        
        action: shape (act_exec_horizon, action_dim)
        """
        total_reward = 0.0
        
        for i in range(self.act_exec_horizon):
            obs, reward, done, truncated, info = self.env.step(action[i])
            total_reward += reward
            
            if done or truncated:
                break
        
        # 更新观测缓冲区
        self.obs_buffer.append(obs)
        
        # 返回观测序列
        stacked_obs = np.stack(list(self.obs_buffer), axis=0)
        
        return stacked_obs, total_reward, done, truncated, info
```

```python
def compute_reward(self, achieved_goal, desired_goal):
    # 1. 距离奖励
    distance = np.linalg.norm(achieved_goal - desired_goal)
    distance_reward = -distance
    
    # 2. 成功奖励
    success = (distance < 0.05)
    success_reward = 10.0 if success else 0.0
```

```python
# my_envs/my_task_env.py

import gym
import numpy as np

class MyTaskEnv(gym.Env):
    """
    自定义任务环境
    """
    
    def __init__(self, **kwargs):
        super().__init__()
        
        # 定义观测空间
        self.observation_space = gym.spaces.Box(
            low=-np.inf,
            high=np.inf,
            shape=(10,),
            dtype=np.float32
        )
        
        # 定义动作空间
        self.action_space = gym.spaces.Box(
            low=-1.0,
            high=1.0,
            shape=(4,),
            dtype=np.float32
        )
        
    def reset(self, seed=None, options=None):
        """重置环境"""
        super().reset(seed=seed)
        
        # 初始化状态
        self.state = np.zeros(10)
        self.step_count = 0
        
        return self.state, {}
    
    def step(self, action):
        """执行一步"""
        # 1. 更新状态
        self.state = self._update_state(action)
        
        # 2. 计算奖励
        reward = self._compute_reward(self.state, action)
        
        # 3. 判断是否终止
        done = self._is_done(self.state)
        truncated = (self.step_count >= 100)
        
        self.step_count += 1
        
        return self.state, reward, done, truncated, {}
    
    def _compute_reward(self, state, action):
        """计算奖励"""
        # 示例：负距离奖励
        goal = np.array([1.0, 1.0, 0.5])
        current_pos = state[:3]
        distance = np.linalg.norm(current_pos - goal)
        
        return -distance
```

```python
class RewardShaper:
    """
    奖励塑形器
    将稀疏奖励转换为稠密奖励
    """
    
    def __init__(self, env):
        self.env = env
        self.prev_distance = None
    
    def __call__(self, obs, action, reward, next_obs, done):
        """
        计算塑形后的奖励
        
        组合多个奖励项:
        1. 任务完成奖励 (稀疏)
        2. 距离改进奖励 (稠密)
        3. 动作平滑奖励 (正则化)
        """
        shaped_reward = 0.0
        
        # 1. 任务完成奖励
        if done and reward > 0:
            shaped_reward += 10.0
        
        # 2. 距离改进奖励
        goal = self.env.get_goal()
        current_pos = next_obs[:3]
        distance = np.linalg.norm(current_pos - goal)
        
        if self.prev_distance is not None:
            improvement = self.prev_distance - distance
            shaped_reward += 5.0 * improvement
        
        self.prev_distance = distance
        
        # 3. 动作平滑奖励
        if hasattr(self, 'prev_action'):
            action_diff = np.linalg.norm(action - self.prev_action)
            shaped_reward -= 0.1 * action_diff
        
        self.prev_action = action
        
        return shaped_reward

# 使用方式
reward_shaper = RewardShaper(env)
shaped_reward = reward_shaper(obs, action, reward, next_obs, done)
```

```python
import flax.linen as nn
import jax.numpy as jnp

class CustomEncoder(nn.Module):
    """
    自定义视觉编码器
    结合 CNN 和 Transformer
    """
    features: Sequence[int] = (32, 64, 128)
    num_heads: int = 4
    
    @nn.compact
    def __call__(self, x: jnp.ndarray):
        # 1. CNN 特征提取
        for feat in self.features:
            x = nn.Conv(feat, kernel_size=(3, 3))(x)
            x = nn.relu(x)
            x = nn.max_pool(x, window_shape=(2, 2), strides=(2, 2))
        
        # 2. Flatten
        batch_size = x.shape[0]
        x = x.reshape(batch_size, -1, x.shape[-1])
        
        # 3. Self-Attention
        x = nn.SelfAttention(num_heads=self.num_heads)(x)
        
        # 4. Global pooling
        x = jnp.mean(x, axis=1)
        
        return x
```

```python
from serl_launcher.agents.continuous.sac import SACAgent

class MyAlgorithmAgent(SACAgent):
    """
    继承 SAC 并添加自定义功能
    """
    
    @partial(jax.jit, static_argnames=['self', 'utd_ratio'])
    def update(self, batch: Batch, utd_ratio: int):
        """
        自定义更新逻辑
        """
        # 1. 调用父类的 SAC 更新
        new_agent, info = super().update(batch, utd_ratio)
        
        # 2. 添加自定义更新
        # 例如：辅助任务学习
        aux_loss = self._compute_auxiliary_loss(batch)
        new_agent = new_agent._update_auxiliary(aux_loss)
        
        info['aux_loss'] = aux_loss
        
        return new_agent, info
    
    def _compute_auxiliary_loss(self, batch):
        """
        计算辅助损失
        例如：状态重构、对比学习等
        """
        # 示例：状态预测损失
        predicted_next_state = self.dynamics_model(
            batch['observations'], batch['actions']
        )
        
        aux_loss = jnp.mean(
            (predicted_next_state - batch['next_observations'])**2
        )
        
        return aux_loss
```

```python
# 使用低精度训练
from jax import config
config.update("jax_enable_x64", False)  # 使用 float32

# 梯度累积（减少显存）
def gradient_accumulation_update(agent, replay_buffer, num_steps=4):
    """
    梯度累积更新
    """
    accumulated_grads = None
    
    for _ in range(num_steps):
        batch = replay_buffer.sample(batch_size // num_steps)
        
        grads = jax.grad(agent.loss_fn)(agent.params, batch)
        
        if accumulated_grads is None:
            accumulated_grads = grads
        else:
            accumulated_grads = jax.tree_map(
                lambda x, y: x + y, accumulated_grads, grads
            )
    
    # 平均梯度
    accumulated_grads = jax.tree_map(
        lambda x: x / num_steps, accumulated_grads
    )
    
    # 应用更新
    new_params = agent.optimizer.apply(accumulated_grads, agent.params)
    
    return agent.replace(params=new_params)
```
