## 🎯 使用方式

### 创建环境

```python
from mujoco_env.mujoco_env.envs.env_factory import make_sim_env, make_real_env, make_env

# 方式1: 使用专门的工厂函数
# 创建仿真环境
sim_env = make_sim_env(
    robot_name="franka_panda",
    task_name="pcb_insertion",
    controller_type="joint_position",
    control_freq=20,
    render_mode="rgb_array"
)

# 创建真机环境
real_env = make_real_env(
    robot_name="franka_panda",
    task_name="pcb_insertion",
    robot_ip="192.168.1.1",
    server_ip="127.0.0.1",
    control_freq=20
)

# 方式2: 使用统一接口
# 仿真模式
env = make_env("franka_panda", "pcb_insertion", mode="sim")

# 真机模式
env = make_env(
    "franka_panda", "pcb_insertion", 
    mode="real",
    robot_ip="192.168.1.1"
)

# 使用方式完全相同
obs, info = env.reset()
for _ in range(100):
    action = env.action_space.sample()
    obs, reward, terminated, truncated, info = env.step(action)
    if terminated or truncated:
        break
env.close()
```

**优势**:
- ✅ 简洁易用
- ✅ 仿真和真机接口统一
- ✅ 参数清晰
- ✅ 易于切换模式

## 🔌 Controller集成

```python
# SimulationRobotEnv 中
def __init__(self, ..., controller_type: str, ...):
    # ...
    # 初始化MuJoCo
    self.model = mujoco.MjModel.from_xml_path(xml_path)
    self.data = mujoco.MjData(self.model)
    
    # 创建controller
    self.controller = get_controller(
        controller_type,
        model=self.model,
        data=self.data,
        dof=self.robot_config.dof,
        control_freq=control_freq
    )

def _set_action(self, action: np.ndarray):
    """使用controller计算控制信号"""
    ctrl_signal = self.controller.compute_control(
        action=action,
        qpos=self.data.qpos[:self.robot_config.dof],
        qvel=self.data.qvel[:self.robot_config.dof],
        model=self.model,
        data=self.data
    )
    self.data.ctrl[:self.robot_config.dof] = ctrl_signal
```

**支持的控制器类型**:
- ✅ `joint_position` - 关节位置控制
- ✅ `joint_velocity` - 关节速度控制
- ✅ `joint_torque` - 关节力矩控制
- ✅ `osc` - 操作空间控制
- ✅ `impedance` - 阻抗控制
- ✅ 易于扩展新的控制器

## 🤖 真机环境集成

```python
# 创建真机环境
env = make_real_env(
    robot_name="franka_panda",
    task_name="pcb_insertion",
    robot_ip="192.168.1.1",
    server_ip="127.0.0.1"
)

# ✅ 标准的Gymnasium环境接口
obs, info = env.reset()
action = env.action_space.sample()
obs, reward, terminated, truncated, info = env.step(action)

# ✅ 与仿真环境使用方式完全相同
# ✅ 可以直接用于RL训练
# ✅ 可以与仿真环境无缝切换
```

**内部实现**:
```python
class RealRobotEnv(BaseRobotEnv):
    def __init__(self, robot_interface, task, ...):
        self.robot_interface = robot_interface  # 组合方式
        self.task = task
        self.robot_interface.connect()  # 连接真机
    
    def step(self, action):
        self._set_action(action)  # 发送到真机
        obs = self._get_obs()     # 从真机获取
        reward = self.task.compute_reward(...)  # 计算奖励
        return obs, reward, terminated, truncated, info
```
